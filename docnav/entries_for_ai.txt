def x_pi_over_4(max_iter: int, seeds: tuple, istep: int = 2) -> float:
 N, D, C = seeds
 d_step = istep**2

 for _ in range(max_iter):
 N = N - (1.0 / D) + (1.0 / C)
 D += d_step
 C += c_step
 return N

 N, D, C = seeds
 for _ in range(max_iter):
 N = 1.0 + (1.0 / N)
